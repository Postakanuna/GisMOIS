 46803  export function getFieldLabel(technicalName: string): string {
 46804    const label = fieldLabels[technicalName] ?? fieldLabelsLower[technicalName.toLowerCase()];
 46805    if (label) {
 46806      return `${label} (${technicalName})`;
 46807    }
 46808    return technicalName;
 46809  }

 46811  export function transformPropertyKeys(
 46812    properties: Record<string, unknown>
 46813  ): Record<string, unknown> {
 46814    const result: Record<string, unknown> = {};
 46815    for (const [key, value] of Object.entries(properties)) {
 46816      const newKey = getFieldLabel(key);
 46817      result[newKey] = value;
 46818    }
 46819    return result;
 46820  }
