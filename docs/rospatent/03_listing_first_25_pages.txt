   288          </div>

   292              <div
   293                key={msg.id}

   309                  {msg.content}
   310                </div>

   312                  <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-muted">

   350              onClick={handleSend}

   355            </Button>

   541                    <Alert className="border-yellow-500 bg-yellow-50 dark:bg-yellow-950">

  1235            newMap.delete(lastAction.featureId);

  1237            newMap.set(lastAction.featureId, {
  1238              ...existingPending,
  1239              newProperties: lastAction.originalProperties,
  1240            });
  1241          }
  1242        }
  1243        return newMap;
  1244      });
