   127	export function AiChatPanel({ onBack, messages, onMessagesChange }: AiChatPanelProps) {
   128	  const [input, setInput] = useState("");
   129	  const [isLoading, setIsLoading] = useState(false);
   130	  const [providers, setProviders] = useState<AiProvider[]>([]);
   131	  const [selectedProvider, setSelectedProvider] = useState("openai");
   132	  const scrollRef = useRef<HTMLDivElement>(null);
   133	  const textareaRef = useRef<HTMLTextAreaElement>(null);
   134	
   135	  useEffect(() => {
   136	    fetch("/api/ai/providers")
   137	      .then(r => r.json())
   138	      .then(data => {
   139	        if (data.providers) setProviders(data.providers);
   140	        if (data.default) setSelectedProvider(data.default);
   141	      })
   142	      .catch(() => {});
   143	  }, []);
   144	
   145	  useEffect(() => {
   146	    if (scrollRef.current) {
   147	      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
   148	    }
   149	  }, [messages]);
   150	
   151	  const handleSend = async () => {
   152	    const text = input.trim();
   153	    if (!text || isLoading) return;
   154	
   155	    const userMsg: ChatMessage = {
   156	      id: `user-${Date.now()}`,
   157	      role: "user",
   158	      content: text,
   159	      timestamp: new Date(),
   160	    };
   161	    const updatedMessages = [...messages, userMsg];
   162	    onMessagesChange(updatedMessages);
   163	    setInput("");
   164	    setIsLoading(true);
   165	
   166	    try {
   167	      const allMessages = updatedMessages.filter(m => m.id !== "welcome");
   168	      const apiMessages = allMessages.map(m => ({ role: m.role, content: m.content }));
   169	
   170	      const response = await fetch("/api/ai/chat", {
   171	        method: "POST",
   172	        headers: { "Content-Type": "application/json" },
   173	        body: JSON.stringify({ messages: apiMessages, provider: selectedProvider }),
   174	      });
   175	      const data = await response.json();
   176	      if (!response.ok) {
   177	        throw new Error(data.error || `Ошибка сервера: ${response.status}`);
   178	      }
   179	
   180	      const aiMsg: ChatMessage = {
   181	        id: `ai-${Date.now()}`,
   182	        role: "assistant",
   183	        content: data.content || "Нет ответа от модели",
   184	        timestamp: new Date(),
   185	      };
   186	      onMessagesChange([...updatedMessages, aiMsg]);
   187	    } catch (error: any) {
   188	      const errorMsg: ChatMessage = {
   189	        id: `error-${Date.now()}`,
   190	        role: "assistant",
   191	        content: `Ошибка: ${error.message || "Не удалось получить ответ от ИИ"}`,
   192	        timestamp: new Date(),
   193	      };
   194	      onMessagesChange([...updatedMessages, errorMsg]);
   195	    } finally {
   196	      setIsLoading(false);
   197	    }
   198	  };
   199	
   200	  const handleClearChat = () => {
   201	    onMessagesChange([WELCOME_MESSAGE]);
   202	  };
   203	
   204	  const handleKeyDown = (e: React.KeyboardEvent) => {
   205	    if (e.key === "Enter" && !e.shiftKey) {
   206	      e.preventDefault();
   207	      handleSend();
   208	    }
   209	  };
   210	
   211	  const currentProvider = providers.find(p => p.id === selectedProvider);
   212	  const hasHistory = messages.some(m => m.id !== "welcome");
   213	
   214	  return (
   215	    <div className="flex flex-col h-full min-h-0 flex-1">
   216	      <div className="flex items-center gap-2 px-4 py-2 border-b shrink-0">
   217	        <Button size="icon" variant="ghost" onClick={onBack} data-testid="button-back-to-layers">
   218	          <ArrowLeft className="h-4 w-4" />
   219	        </Button>
   220	        <Bot className="h-4 w-4 text-primary" />
   221	        <span className="text-sm font-semibold">ИИ-ассистент</span>
   222	        <div className="ml-auto flex items-center gap-1">
   223	          {hasHistory && (
   224	            <Button
   225	              size="icon"
   226	              variant="ghost"
   227	              onClick={handleClearChat}
   228	              className="h-7 w-7"
   229	              title="Очистить чат"
   230	              data-testid="button-clear-chat"
   231	            >
   232	              <Trash2 className="h-3.5 w-3.5" />
   233	            </Button>
   234	          )}
   235	          <DropdownMenu>
   236	            <DropdownMenuTrigger asChild>
   237	              <Button variant="outline" size="sm" className="h-7 text-xs gap-1 px-2" data-testid="button-provider-selector">
   238	                {currentProvider?.name || "Модель"}
   239	                <ChevronDown className="h-3 w-3" />
   240	              </Button>
   241	            </DropdownMenuTrigger>
   242	            <DropdownMenuContent align="end">
   243	              {providers.map(p => (
   244	                <DropdownMenuItem
   245	                  key={p.id}
   246	                  onClick={() => setSelectedProvider(p.id)}
   247	                  disabled={!p.available}
   248	                  data-testid={`provider-option-${p.id}`}
   249	                >
   250	                  <span className={selectedProvider === p.id ? "font-semibold" : ""}>
   251	                    {p.name}
   252	                  </span>
   253	                  {!p.available && <span className="ml-2 text-muted-foreground text-xs">(не настроен)</span>}
   254	                </DropdownMenuItem>
   255	              ))}
   256	            </DropdownMenuContent>
   257	          </DropdownMenu>
   258	        </div>
   259	      </div>
   260	
   261	      <div ref={scrollRef} className="flex-1 overflow-y-auto p-3 space-y-3 min-h-0">
   262	        {messages.map((msg) => (
   263	          <div
   264	            key={msg.id}
   265	            className={`flex gap-2 ${msg.role === "user" ? "justify-end" : "justify-start"}`}
   266	            data-testid={`chat-message-${msg.id}`}
   267	          >
   268	            {msg.role === "assistant" && (
   269	              <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-primary/10">
   270	                <Bot className="h-3.5 w-3.5 text-primary" />
   271	              </div>
   272	            )}
   273	            <div
   274	              className={`rounded-md px-3 py-2 text-sm max-w-[85%] whitespace-pre-wrap ${
   275	                msg.role === "user"
   276	                  ? "bg-primary text-primary-foreground"
   277	                  : "bg-muted"
   278	              }`}
   279	            >
   280	              {msg.content}
   281	            </div>
   282	            {msg.role === "user" && (
   283	              <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-muted">
   284	                <User className="h-3.5 w-3.5" />
   285	              </div>
   286	            )}
   287	          </div>
   288	        ))}
   289	        {isLoading && (
   290	          <div className="flex gap-2 justify-start" data-testid="chat-loading">
   291	            <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-primary/10">
   292	              <Bot className="h-3.5 w-3.5 text-primary" />
   293	            </div>
   294	            <div className="rounded-md px-3 py-2 text-sm bg-muted">
   295	              <span className="inline-flex gap-1">
   296	                <span className="animate-bounce" style={{ animationDelay: "0ms" }}>.</span>
   297	                <span className="animate-bounce" style={{ animationDelay: "150ms" }}>.</span>
   298	                <span className="animate-bounce" style={{ animationDelay: "300ms" }}>.</span>
   299	              </span>
   300	            </div>
   301	          </div>
   302	        )}
   303	      </div>
   304	
   305	      <div className="border-t p-3 shrink-0">
   306	        <div className="flex gap-2">
   307	          <Textarea
   308	            ref={textareaRef}
   309	            value={input}
   310	            onChange={(e) => setInput(e.target.value)}
   311	            onKeyDown={handleKeyDown}
   312	            placeholder="Задайте вопрос..."
   313	            className="resize-none min-h-[40px] max-h-[120px] text-sm"
   314	            rows={1}
   315	            disabled={isLoading}
   316	            data-testid="input-ai-chat"
   317	          />
   318	          <Button
   319	            size="icon"
   320	            onClick={handleSend}
   321	            disabled={!input.trim() || isLoading}
   322	            data-testid="button-send-ai-chat"
   323	          >
   324	            <Send className="h-4 w-4" />
   325	          </Button>
   326	        </div>
   327	      </div>
   328	    </div>
   329	  );
   330	}

   377	export function ApiKeysManager() {
   378	  const { toast } = useToast();
   379	  const [isCreateOpen, setIsCreateOpen] = useState(false);
   380	  const [newKeyName, setNewKeyName] = useState("");
   381	  const [newKeySceneId, setNewKeySceneId] = useState<string>("");
   382	  const [selectedPermissions, setSelectedPermissions] = useState<Permission[]>(["create_point", "read_scenes", "read_layers"]);
   383	  const [createdToken, setCreatedToken] = useState<string | null>(null);
   384	  const [tokenCopied, setTokenCopied] = useState(false);
   385	
   386	  const { data: apiKeys = [], isLoading: isLoadingKeys } = useQuery<ApiKey[]>({
   387	    queryKey: ["/api/api-keys"],
   388	  });
   389	
   390	  const { data: scenes = [] } = useQuery<Scene[]>({
   391	    queryKey: ["/api/scenes"],
   392	  });
   393	
   394	  const createMutation = useMutation({
   395	    mutationFn: async (data: { name: string; sceneId?: number; permissions: string[] }) => {
   396	      const res = await apiRequest("POST", "/api/api-keys", data);
   397	      return res.json();
   398	    },
   399	    onSuccess: (response) => {
   400	      setCreatedToken(response.token);
   401	      setNewKeyName("");
   402	      setNewKeySceneId("");
   403	      queryClient.invalidateQueries({ queryKey: ["/api/api-keys"] });
   404	      toast({
   405	        title: "API ключ создан",
   406	        description: "Скопируйте токен, он больше не будет показан!",
   407	      });
   408	    },
   409	    onError: (error: Error) => {
   410	      toast({
   411	        title: "Ошибка создания ключа",
   412	        description: error.message,
   413	        variant: "destructive",
   414	      });
   415	    },
   416	  });
   417	
   418	  const revokeMutation = useMutation({
   419	    mutationFn: async (id: number) => {
   420	      const res = await apiRequest("DELETE", `/api/api-keys/${id}`);
   421	      return res.json();
   422	    },
   423	    onSuccess: () => {
   424	      queryClient.invalidateQueries({ queryKey: ["/api/api-keys"] });
   425	      toast({
   426	        title: "API ключ отозван",
   427	        description: "Ключ больше не может использоваться",

   429	    },
   430	    onError: (error: Error) => {
   431	      toast({
   432	        title: "Ошибка отзыва ключа",

   434	        variant: "destructive",
   435	      });
   436	    },

   439	  const handleCreate = () => {

   443	      name: newKeyName.trim(),

   447	  };

   457	  const handleCloseDialog = () => {

   468	        : [...prev, permission]

   470	  };

   476	      month: "2-digit",
   477	      year: "numeric",

   479	      minute: "2-digit",

   481	  };

   489	            <CardTitle>API ключи</CardTitle>
   490	          </div>
   491	          <Dialog open={isCreateOpen} onOpenChange={(open) => {
   492	            if (!open) handleCloseDialog();
   493	            else setIsCreateOpen(true);
   494	          }}>
   495	            <DialogTrigger asChild>
   496	              <Button size="sm" data-testid="button-create-api-key">
   497	                <Plus className="h-4 w-4 mr-1" />
   498	                Создать ключ
   499	              </Button>
   500	            </DialogTrigger>
   501	            <DialogContent className="sm:max-w-md">
   502	              <DialogHeader>
   503	                <DialogTitle>Создать API ключ</DialogTitle>
   504	                <DialogDescription>
   505	                  API ключ позволяет внешним приложениям (например, Telegram бот) создавать точки на карте.
   506	                </DialogDescription>
   507	              </DialogHeader>
   508

   510	                <div className="space-y-4">
   511	                  <Alert className="border-yellow-500 bg-yellow-50 dark:bg-yellow-950">
   512	                    <AlertCircle className="h-4 w-4 text-yellow-600" />

   551	                      onChange={(e) => setNewKeyName(e.target.value)}

   556	                  <div className="space-y-2">
   557	                    <Label htmlFor="api-key-scene">Ограничить сценой (опционально)</Label>
